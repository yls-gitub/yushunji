    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { getAuth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
        import { getFirestore, collection, doc, setDoc, deleteDoc, onSnapshot } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        // 获取 Firebase 全局配置
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'yushunji-app';
        const firebaseConfig = typeof __firebase_config !== 'undefined' 
            ? JSON.parse(__firebase_config) 
            : {
                apiKey: "AIzaSyDemoConfigForYushunJiKey",
                authDomain: "yushunji.firebaseapp.com",
                projectId: "yushunji-demo",
                storageBucket: "yushunji-demo.appspot.com"
            };

        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);

        const { createApp } = Vue;

        const CITY_COORDS = {
            '西安': [34.2658, 108.9541],
            '北京': [39.9042, 116.4074],
            '上海': [31.2304, 121.4737]
        };

        createApp({
            data() {
                return {
                    activeTab: 'inventory',
                    activeModal: null,
                    searchQuery: '',
                    statusFilter: 'in_stock',
                    dynastyFilter: '',
                    ownerFilter: 'all',
                    isCloudConnected: false,
                    user: null,
                    
                    coins: [],
                    selectedCoin: null,
                    map: null,
                    markers: [],

                    formBuy: {
                        name: '',
                        owner: '虞',
                        dynasty: '唐代',
                        material: '黄铜',
                        buyPrice: null,
                        buyDate: new Date().toISOString().substr(0, 10),
                        buyShop: '',
                        buyAddress: '',
                        grade: '',
                        notes: '',
                        imgFront: '',
                        imgBack: ''
                    }
                }
            },
            computed: {
                yuCoins() { return this.coins.filter(c => c.owner === '虞'); },
                shunCoins() { return this.coins.filter(c => (c.owner || '舜') === '舜'); },
                yuTotalCost() { return this.yuCoins.reduce((sum, c) => sum + (Number(c.buyPrice) || 0), 0); },
                shunTotalCost() { return this.shunCoins.reduce((sum, c) => sum + (Number(c.buyPrice) || 0), 0); },
                filteredOwnerCoins() {
                    if (this.ownerFilter === 'all') return this.coins;
                    return this.ownerFilter === '虞' ? this.yuCoins : this.shunCoins;
                },
                locationGroups() {
                    const groups = {};
                    this.coins.forEach(coin => {
                        const addr = coin.buyAddress ? coin.buyAddress.trim() : '其他';
                        if (!groups[addr]) groups[addr] = { coins: [], totalCost: 0 };
                        groups[addr].coins.push(coin);
                        groups[addr].totalCost += (Number(coin.buyPrice) || 0);
                    });
                    return groups;
                },
                dynastyOptions() {
                    return Array.from(new Set(this.coins.map(c => c.dynasty).filter(Boolean)));
                },
                inStockCoins() { return this.coins.filter(c => c.status === 'in_stock'); },
                soldCoins() { return this.coins.filter(c => c.status === 'sold'); },
                filteredCoins() {
                    return this.coins.filter(coin => {
                        if (this.ownerFilter !== 'all' && (coin.owner || '舜') !== this.ownerFilter) return false;
                        if (this.statusFilter === 'in_stock' && coin.status !== 'in_stock') return false;
                        if (this.statusFilter === 'sold' && coin.status !== 'sold') return false;
                        if (this.dynastyFilter && coin.dynasty !== this.dynastyFilter) return false;
                        if (this.searchQuery) {
                            const q = this.searchQuery.toLowerCase();
                            return coin.name && coin.name.toLowerCase().includes(q);
                        }
                        return true;
                    });
                },
                stats() {
                    let totalInvestment = 0, totalSales = 0, realizedProfit = 0;
                    this.coins.forEach(c => {
                        const buyP = Number(c.buyPrice) || 0;
                        totalInvestment += buyP;
                        if (c.status === 'sold') {
                            const sellP = Number(c.sellPrice) || 0;
                            totalSales += sellP;
                            realizedProfit += (sellP - buyP);
                        }
                    });
                    return { totalInvestment, totalSales, realizedProfit, inStockCount: this.inStockCoins.length };
                }
            },
            async mounted() {
                // 初始化云端协同认证与实时数据监听
                await this.initCloudSync();
            },
            methods: {
                // 初始化 Firestore 协同实时同步机制
                async initCloudSync() {
                    try {
                        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                            await signInWithCustomToken(auth, __initial_auth_token);
                        } else {
                            await signInAnonymously(auth);
                        }

                        onAuthStateChanged(auth, (usr) => {
                            if (usr) {
                                this.user = usr;
                                this.listenToCloudCoins();
                            }
                        });
                    } catch (e) {
                        console.warn("使用本地离线存储回退:", e);
                        this.loadLocalCoins();
                    }
                },

                // 监听云端 Firestore 数据变化 (即时协同同步)
                listenToCloudCoins() {
                    if (!this.user) return;
                    // 使用标准的规范路径 /artifacts/{appId}/public/data/coins
                    const coinsCol = collection(db, 'artifacts', appId, 'public', 'data', 'coins');

                    onSnapshot(coinsCol, (snapshot) => {
                        const cloudCoins = [];
                        snapshot.forEach((doc) => {
                            cloudCoins.push({ id: doc.id, ...doc.data() });
                        });
                        this.coins = cloudCoins;
                        this.isCloudConnected = true;
                    }, (error) => {
                        console.error("实时同步出错:", error);
                        this.loadLocalCoins();
                    });
                },

                loadLocalCoins() {
                    const localData = localStorage.getItem('gudong_coins_data');
                    if (localData) this.coins = JSON.parse(localData);
                },

                switchTab(tab) {
                    this.activeTab = tab;
                    if (tab === 'map') this.$nextTick(() => this.initMap());
                },

                initMap() {
                    if (!this.map) {
                        this.map = L.map('buyMap').setView([34.3416, 108.9398], 5);
                        L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png').addTo(this.map);
                    }
                },

                formatNum(val) {
                    return Number(val || 0).toLocaleString('zh-CN', { maximumFractionDigits: 2 });
                },

                openModal(type, coin = null) {
                    this.activeModal = type;
                    if (coin) this.selectedCoin = coin;
                },

                closeModal() {
                    this.activeModal = null;
                },

                handleImageUpload(event, key) {
                    const file = event.target.files[0];
                    if (!file) return;
                    const reader = new FileReader();
                    reader.onload = (e) => {
                        this.formBuy[key] = e.target.result;
                    };
                    reader.readAsDataURL(file);
                },

                // 保存买入记录至云端
                async saveBuyRecord() {
                    const docId = 'coin_' + Date.now();
                    const newCoin = { ...this.formBuy, status: 'in_stock', createdAt: new Date().toISOString() };

                    if (this.isCloudConnected) {
                        const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'coins', docId);
                        await setDoc(docRef, newCoin);
                    } else {
                        this.coins.unshift({ id: docId, ...newCoin });
                        localStorage.setItem('gudong_coins_data', JSON.stringify(this.coins));
                    }
                    this.closeModal();
                },

                // 从云端删除记录
                async deleteCoin(id) {
                    if (confirm('确定要在所有协同设备上同步删除此藏品吗？')) {
                        if (this.isCloudConnected) {
                            const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'coins', id);
                            await deleteDoc(docRef);
                        } else {
                            this.coins = this.coins.filter(c => c.id !== id);
                            localStorage.setItem('gudong_coins_data', JSON.stringify(this.coins));
                        }
                        this.closeModal();
                    }
                },

                getCoinProfit(coin) {
                    return (Number(coin.sellPrice) || 0) - (Number(coin.buyPrice) || 0);
                }
            }
        }).mount('#app');
    </script>
